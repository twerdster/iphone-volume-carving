cmake_minimum_required(VERSION 3.22)
project(volumecarving_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set_property(GLOBAL PROPERTY JOB_POOLS carve_compile=2 carve_link=1)
set(CMAKE_JOB_POOL_COMPILE carve_compile)
set(CMAKE_JOB_POOL_LINK carve_link)
set(OpenCV_STATIC ON)
set(WVC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
get_filename_component(UPDATED "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
add_subdirectory("${UPDATED}/native" "${CMAKE_CURRENT_BINARY_DIR}/recovered-native")
add_library(carving_android SHARED bridge.cpp)
target_compile_definitions(carving_android PRIVATE WVC_VOLUME=1)
target_link_libraries(carving_android PRIVATE volume_carving_native log)
